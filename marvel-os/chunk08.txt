function exportMarvelData(){

 Object.keys(localStorage).filter(k=>k.startsWith("mtos")).forEach(k=>data.storage[k]=localStorage.getItem(k));

 const a=document.createElement("a");a.href=URL.createObjectURL(blob);a.download="marvel-os-backup-"+localDateISO()+".json";document.body.appendChild(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(a.href),1000);

}
function importMarvelData(file){
 const r=new FileReader();r.onload=()=>{
  try{const data=JSON.parse(r.result);if(!data.storage)throw new Error("Invalid backup");Object.entries(data.storage).forEach(([k,v])=>localStorage.setItem(k,v));if(qs("#backupStatus"))qs("#backupStatus").textContent="Backup imported. Reloading…";setTimeout(()=>location.reload(),500)}
  catch(e){if(qs("#backupStatus"))qs("#backupStatus").textContent="Could not import this backup."}
 };r.readAsText(file);
}
