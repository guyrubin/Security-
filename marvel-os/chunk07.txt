function historicalSetLogs(){
 const keys=Object.keys(localStorage).filter(k=>k.startsWith("mtosSets:")).sort();
 return keys.map(k=>({date:k.replace("mtosSets:",""),data:JSON.parse(localStorage.getItem(k)||"{}")}));

function lastSessionForExercise(exId,beforeDate=localDateISO()){
 const logs=historicalSetLogs().filter(x=>x.date<beforeDate&&x.data[exId]&&x.data[exId].some(s=>s.kg||s.reps));
 return logs.length?logs.at(-1):null;
}
function beatTargetText(exId){
 const last=lastSessionForExercise(exId);if(!last)return "First logged session — establish a clean baseline.";
 const sets=last.data[exId], valid=sets.filter(s=>+s.reps>0), total=valid.reduce((a,s)=>a+(+s.reps||0),0);
 const maxKg=Math.max(0,...valid.map(s=>+s.kg||0));
 return maxKg?`Last: ${maxKg}kg • ${total} total reps. Beat the total cleanly before adding load.`:`Last: ${total} total reps. Beat the total with clean form.`;
}
